% PREAMBLE COMMUN - Certification ECNSA
% ------------------------------------------------------------------------------
% Ce fichier ne contient pas de texte : il ne fait que charger des paquets et
% régler la mise en page. Il doit être appelé APRÈS \documentclass et AVANT
% \begin{document}.
%
% Rappel : un « paquet » (package) est un fichier de commandes LaTeX fourni par
% une distribution (TeX Live, MiKTeX...). \usepackage{nom} active ses commandes.
% ==============================================================================

% ------------------------------------------------------------------------------
% 1. ENCODAGE ET LANGUE
% ------------------------------------------------------------------------------

% inputenc : indique à LaTeX comment lire le fichier source.
% L'option [utf8] permet d'écrire directement les accents (é, à, ç...).
% Sans ce paquet, les caractères accentués peuvent produire des erreurs.
\usepackage[utf8]{inputenc}

% fontenc : choisit l'encodage des polices de sortie.
% L'option [T1] améliore le rendu des accents et le copier-coller depuis le PDF.
\usepackage[T1]{fontenc}

% babel : applique les règles typographiques d'une langue.
% L'option [french] gère les espaces avant : ; ! ?, les guillemets français, etc.
\usepackage[french]{babel}

% lmodern : charge la famille de polices Latin Modern.
% Elle est plus lisible et plus complète que la police par défaut.
\usepackage{lmodern}

% ------------------------------------------------------------------------------
% 2. MISE EN PAGE (GEOMETRY)
% ------------------------------------------------------------------------------

% geometry : règle les marges et le format du papier.
% [a4paper] impose le format A4.
% [margin=2.2cm] fixe les quatre marges (haut, bas, gauche, droite) à 2,2 cm.
\usepackage[a4paper, margin=2.2cm, headheight=24pt, headsep=0.5cm, footskip=0.8cm, includeheadfoot]{geometry}

% ------------------------------------------------------------------------------
% 3. TABLEAUX
% ------------------------------------------------------------------------------

% booktabs : fournit les traits horizontaux professionnels des tableaux :
%   \toprule    -> trait épais du haut
%   \midrule    -> trait fin après l'en-tête
%   \bottomrule -> trait épais du bas
% Il faut éviter les traits verticaux avec ce style.
\usepackage{booktabs}

% array : améliore la définition des colonnes de tableau.
% Il permet d'appliquer un format à une colonne, par exemple :
%   >{\bfseries}p{3cm}  -> colonne de 3 cm en gras
%   >{\raggedright\arraybackslash}p{3cm} -> colonne de 3 cm alignée à gauche
\usepackage{array}

% tabularx : crée des tableaux qui occupent toute la largeur imposée.
% Sa colonne spéciale X se partage automatiquement l'espace restant.
% Syntaxe : \begin{tabularx}{\linewidth}{p{3cm} X} ... \end{tabularx}
\usepackage{tabularx}

% longtable : crée un tableau qui peut continuer sur plusieurs pages.
% On y répète l'en-tête avec \endhead après la première ligne d'en-tête.
\usepackage{longtable}
\usepackage{pdflscape}

% multirow : permet de fusionner plusieurs lignes dans une même cellule.
% Syntaxe : \multirow{nombre_de_lignes}{largeur}{texte}
\usepackage{multirow}

% ------------------------------------------------------------------------------
% 4. IMAGES ET COULEURS
% ------------------------------------------------------------------------------

% graphicx : charge les images avec \includegraphics.
% Options utiles : width=0.8\linewidth, height=5cm, keepaspectratio.
\usepackage{graphicx}

% float : fournit l'option [H] pour les environnements figure et table.
% [H] signifie « ici, exactement », au lieu de laisser LaTeX déplacer l'objet.
\usepackage{float}

% xcolor : permet d'utiliser des couleurs (\textcolor, \colorbox...).
% Il est chargé pour les cas où un document doit mettre en évidence un texte.
\usepackage[table]{xcolor}

% caption : personnalise les légendes des figures et des tableaux.
\usepackage{caption}
\usepackage{microtype}
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable}

% ------------------------------------------------------------------------------
% 5. LISTES ET ESPACEMENTS
% ------------------------------------------------------------------------------

% enumitem : règle finement les listes itemize et enumerate.
% Exemple : \begin{itemize}[leftmargin=*, itemsep=2pt]
\usepackage{enumitem}

% \parindent : retrait de première ligne d'un paragraphe.
% 0pt supprime l'alinéa, style courant des documents techniques.
\setlength{\parindent}{0pt}

% \parskip : espace vertical ajouté entre deux paragraphes.
% 0.5em vaut environ la moitié d'une ligne : le texte est plus aéré.
\setlength{\parskip}{0.5em}

% ------------------------------------------------------------------------------
% 6. LIENS ET NAVIGATION
% ------------------------------------------------------------------------------

% hyperref : rend cliquables les liens, le sommaire et les renvois.
% [hidelinks] supprime les cadres colorés autour des liens.
% Il doit être chargé en dernier, sauf exception documentée.
\usepackage[hidelinks]{hyperref}

% titlesec : personnalise les titres de sections et sous-sections.
\usepackage{titlesec}

% ------------------------------------------------------------------------------
% 7. RÉGLAGES GLOBAUX
% ------------------------------------------------------------------------------

% \arraystretch : hauteur des lignes de tableau.
% 1.25 ajoute 25 % de hauteur pour éviter que le texte touche les traits.
\renewcommand{\arraystretch}{1.25}

% \captionsetup : réglage des légendes.
% font=small      -> légende en petite taille
% labelfont=bf    -> mot « Figure » ou « Tableau » en gras
\captionsetup{font=small, labelfont=bf}

% ------------------------------------------------------------------------------
% 8. IDENTITÉ VISUELLE ET AIDES À LA RÉDACTION
% ------------------------------------------------------------------------------

% À appeler après le préambule propre au type de document.
% #1 est le nom du document, conservé pour compatibilité avec les profils.
% #2 est sa couleur d'accent.
% \pagestyle{plain} : numérotation centrée en bas, sans en-tête ni pied personnalisé.
\newcommand{\configurerdocument}[2]{%
	\colorlet{accentdocument}{#2}%
	\pagestyle{plain}%
	\titleformat{\section}{\Large\bfseries\color{bleunuit}}{\thesection.}{0.6em}{}%
	\titleformat{\subsection}{\large\bfseries\color{bleunuit}}{\thesubsection}{0.6em}{}%
	\titlespacing*{\section}{0pt}{2.0em}{0.8em}%
	\titlespacing*{\subsection}{0pt}{1.3em}{0.45em}%
}

% Page de garde prête à remplir : type, titre, sous-titre, auteur(s), date.
\newcommand{\fairepagegarde}[5]{%
	\begin{titlepage}
		\thispagestyle{empty}
		\color{bleunuit}
		\vspace*{1.2cm}
		{\color{accentdocument}\rule{\linewidth}{3pt}}\par
		\vspace{1.1cm}
		{\Large\bfseries\textcolor{accentdocument}{#1}\par}
		\vspace{0.8cm}
		{\Huge\bfseries #2\par}
		\vspace{0.5cm}
		{\Large\color{gray}#3\par}
		\vspace{1.5cm}
		\begin{tcolorbox}[enhanced,breakable,colback=grisclair,colframe=grisclair,
			borderline west={3pt}{0pt}{accentdocument},boxrule=0pt,arc=1mm,
			left=5mm,right=5mm,top=4mm,bottom=4mm]
			\begin{tabularx}{\linewidth}{>{\bfseries}p{2.6cm} X}
				Auteur(s) & #4 \\
				Formation & \codeformation \\
				Chapitre & \chapitrecours \\
				\if\relax\detokenize\expandafter{\liengit}\relax\else Dépôt Git & \url{\liengit} \\ \fi
				Date & #5
			\end{tabularx}
		\end{tcolorbox}
	\end{titlepage}%
}

% Encadré reproductible pour un filtre, une commande ou une méthode d'analyse.
\newcommand{\methodeanalyse}[2]{%
	\begin{blocpedagogique}{#1}
		\texttt{\detokenize{#2}}%
	\end{blocpedagogique}%
}

% Capture légendée avec nom de fichier et filtre/commande associés.
% Arguments : identifiant de label, fichier image, filtre/commande, légende.
\newcommand{\captureanalyse}[4]{%
	\begin{figure}[htbp]
		\centering
		\IfFileExists{#2}{%
			\includegraphics[width=0.9\linewidth,height=7cm,keepaspectratio]{#2}%
		}{%
			\fbox{\parbox{0.82\linewidth}{\centering
				Capture à ajouter : \texttt{\detokenize{#2}}\\
				Recadrer sur l'élément utile et masquer les données sensibles.}}%
		}
		\caption{#4}
		\label{#1}
		\begin{minipage}{0.9\linewidth}
			\footnotesize\textcolor{gray}{\textbf{Fichier :} \texttt{\detokenize{#2}}\quad
			\textbf{Filtre / commande :} \texttt{\detokenize{#3}}}
		\end{minipage}
	\end{figure}%
}
